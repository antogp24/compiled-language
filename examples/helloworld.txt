struct Rect2D {
    top_left: vec2,
    size: vec2,
}

struct Player {
    collider: Rect2D,
    hp: i32,
    texture_id: u32,
}

fn damage(self: *Player, value: i32) -> bool {
    self.hp -= value;
    self.hp = max(self.hp, 0);
    return self.hp > 0;
}

let players: [2]Player = {
    { .hp = 65 },
    { .hp = 1 },
};

fn main() {
    if damage(&players[0], 5) {
        println("Player 1 is still alive");
    } else if damage(&players[1], 5) {
        println("Player 2 is still alive");
    } else {
        println("All players are dead.");
    }
	exit(0);
}
