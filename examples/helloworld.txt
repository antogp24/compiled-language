struct Rect2D {
    top_left: vec2,
    size: vec2,
}

struct Player {
    collider: Rect2D,
    hp: i32,
    texture_id: u32,
}

fn damage(self: *Player, value: int) -> bool {
    self.hp -= value;
    self.hp = max(self.hp, 0);
    return self.hp > 0;
}

let players: [1]Player = {
    { .hp = 65 },
};

fn main() {
    if damage(&players[0], 5) {
        println("Player is alive");
        exit(0);
    } else {
        println("Player was killed");
        exit(1);
    }
}
